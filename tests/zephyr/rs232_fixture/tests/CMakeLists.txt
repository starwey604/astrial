cmake_minimum_required(VERSION 3.22)
project(astrial_rs232_fixture_tests LANGUAGES C)
enable_testing()
option(FIXTURE_SANITIZERS "Enable address and undefined-behaviour sanitizers" OFF)
add_library(fixture_core ../src/fixture.c)
target_include_directories(fixture_core PUBLIC ../src)
target_compile_features(fixture_core PUBLIC c_std_11)
target_compile_options(fixture_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(fixture_test fixture_test.c)
target_link_libraries(fixture_test PRIVATE fixture_core)
target_compile_options(fixture_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
if(FIXTURE_SANITIZERS)
  foreach(t fixture_core fixture_test)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(fixture_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME rs232_fixture_core COMMAND fixture_test)
